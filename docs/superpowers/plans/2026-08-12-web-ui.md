# Read-Only Web UI Dashboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A minimal, server-rendered web dashboard (portfolio, pools, agent history) for vexis-dlmm-bot, running as a separate process with password login — read-only.

**Architecture:** New `src/web/` module using `@effect/platform` `HttpServer`/`HttpRouter` served by `NodeHttpServer.layer`. Pages are server-rendered HTML with htmx (CDN) auto-refresh. Data comes from the existing `AppLayer` services (`MeteoraApi`, `Dlmm`, `Screening`, `AppConfig`) — zero changes to existing service code. Auth = password → HMAC-signed session cookie.

**Tech Stack:** Effect 3.22 (`@effect/platform` 0.97, `@effect/platform-node` 0.108 — both already installed), TypeScript strict ESM, vitest, htmx 1.9.x via CDN, `node:crypto` for auth.

## Global Constraints

- ESM-only — every import of a local file uses the `.js` extension (`import { x } from "./foo.js"`)
- Biome formatting: tab indent, double quotes; run `npm run check` (biome check --write) before finishing each task
- TypeScript strict, no unused locals/params; `npm run typecheck` must pass
- Vitest unit tests only — no live API calls, all fixtures inline
- No new npm dependencies (htmx is loaded from CDN in the HTML shell)
- All dynamic values rendered into HTML must pass through `escapeHtml` from `src/web/layout.ts`
- Tagged errors from `src/errors.ts`, not thrown exceptions (except `console`-level fatal handling in entry points)
- Config additions to `VexisConfig` are plain TypeScript interfaces (matching existing `src/domain/config.ts` style — it is NOT Effect.Schema, unlike API-response types)

Verified APIs (from installed `@effect/platform@0.97` / `@effect/platform-node@0.108`):

```ts

NodeHttpServer.layer(() => createServer(), { port })  // Layer providing @effect/platform/HttpServer
```

```ts
export function resolveWebConfig(
	cfg: VexisConfig,
	env: Record<string, string | undefined> = process.env,

	return {

		port: web.port ?? 8080,
```

Create `src/web/auth.ts`:

```ts
import { createHmac, timingSafeEqual } from "node:crypto";

export const SESSION_COOKIE = "vexis_session";
export const SESSION_TTL_MS = 24 * 60 * 60 * 1000;

function hmac(password: string, payload: string): Buffer {
	return createHmac("sha256", password).update(payload).digest();
}

export function passwordMatches(input: string, expected: string): boolean {
	const a = Buffer.from(input, "utf8");
	const b = Buffer.from(expected, "utf8");
	if (a.length !== b.length) return false;
	return timingSafeEqual(a, b);
}

export function signSessionCookie(password: string, now = Date.now()): string {
	const payload = Buffer.from(
		JSON.stringify({ exp: now + SESSION_TTL_MS }),
		"utf8",
	).toString("base64url");
	return `${payload}.${hmac(password, payload).toString("hex")}`;
}

export function verifySessionCookie(
	cookieValue: string,
	password: string,
	now = Date.now(),
): boolean {
	const dot = cookieValue.lastIndexOf(".");
	if (dot <= 0) return false;
	const payload = cookieValue.slice(0, dot);
	const given = Buffer.from(cookieValue.slice(dot + 1), "hex");
	const expected = hmac(password, payload);
	if (given.length !== expected.length || !timingSafeEqual(given, expected)) {
		return false;
	}
	try {
		const parsed = JSON.parse(
			Buffer.from(payload, "base64url").toString("utf8"),
		) as { exp?: unknown };
		if (typeof parsed.exp !== "number" || parsed.exp <= now) return false;
		return true;
	} catch {
		return false;
	}
}

export function sessionCookieHeader(password: string): string {
	return `${SESSION_COOKIE}=${signSessionCookie(password)}; Path=/; HttpOnly; SameSite=Lax; Max-Age=${Math.floor(SESSION_TTL_MS / 1000)}`;
}

export function expiredCookieHeader(): string {
	return `${SESSION_COOKIE}=; Path=/; HttpOnly; SameSite=Lax; Max-Age=0`;
}
```

Note: `"a.b"` → payload `"a"`, signature `"b"` → `Buffer.from("b", "hex")` is empty (length 0 ≠ 32) → `false`. Tampered signature → 32 bytes but different value → `timingSafeEqual` false. Expired → `exp <= now` → false.

```ts
import { formatNum } from "../format.js";
import { escapeHtml } from "./layout.js";

export function fmtUsd(v: string | number | null | undefined): string {
	if (v === null || v === undefined) return "-";
	const n = typeof v === "number" ? v : parseFloat(v);
	if (Number.isNaN(n)) return "-";
	return `$${formatNum(n)}`;
}

export function fmtPct(v: string | number | null | undefined): string {
	if (v === null || v === undefined) return "-";
	const n = typeof v === "number" ? v : parseFloat(v);
	if (Number.isNaN(n)) return "-";
	const sign = n > 0 ? "+" : "";
	return `${sign}${formatNum(n)}%`;
}

export function fmtSol(v: string | number | null | undefined): string {
	if (v === null || v === undefined) return "-";
	const n = typeof v === "number" ? v : parseFloat(v);
	if (Number.isNaN(n)) return "-";
	return `${formatNum(n, 3)} ◎`;
}

export function pnlClass(v: number): "pos" | "neg" | "zero" {
	return v > 0 ? "pos" : v < 0 ? "neg" : "zero";
}

export type BadgeKind = "ok" | "warn" | "danger" | "neutral";

export function badge(text: string, kind: BadgeKind): string {
	return `<span class="badge ${kind}">${escapeHtml(text)}</span>`;
}

export function summaryCard(label: string, value: string, sub: string): string {
	return `<div class="card"><div class="label">${escapeHtml(label)}</div><div class="value">${value}</div><div class="sub">${sub}</div></div>`;
}

export function table(headers: readonly string[], rows: readonly string[]): string {
	const head = headers.map((h) => `<th>${escapeHtml(h)}</th>`).join("");
	return `<table><thead><tr>${head}</tr></thead><tbody>\n${rows.join("\n")}\n</tbody></table>`;
}

export function sparkline(values: readonly number[], width = 200, height = 40): string {
	if (values.length < 2) return "";
	const min = Math.min(...values);
	const max = Math.max(...values);
	const span = max - min || 1;
	const points = values
		.map((v, i) => {
			const x = (i / (values.length - 1)) * width;
			const y = height - ((v - min) / span) * height;
			return `${x.toFixed(1)},${y.toFixed(1)}`;
		})
		.join(" ");
	return `<svg viewBox="0 0 ${width} ${height}" width="${width}" height="${height}" role="img" aria-label="trend"><polyline points="${points}" fill="none" stroke="currentColor" stroke-width="1.5"/></svg>`;
}

export function meteoraUrl(pool: string): string {
	return `https://app.meteora.ag/dlmm/${pool}`;
}

export function solscanUrl(sig: string): string {
	return `https://solscan.io/tx/${sig}`;
}

export function tsLocal(ts: string | number | null | undefined): string {
	if (ts === null || ts === undefined) return "-";
	const ms = typeof ts === "number" ? ts * 1000 : Date.parse(ts);
	if (Number.isNaN(ms)) return "-";
	const d = new Date(ms);
	const p = (n: number) => String(n).padStart(2, "0");
	return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`;
}
```

```ts
export interface PortfolioData {
	readonly total: PortfolioTotal;
	readonly open: readonly OpenPool[];
	readonly closed: readonly ClosedPool[];
}

export function renderPortfolio(data: PortfolioData): string {
	const total = data.total;
	const openBalance = data.open.reduce((s, p) => s + parseFloat(p.balances || "0"), 0);
	const openFees = data.open.reduce((s, p) => s + parseFloat(p.unclaimedFees || "0"), 0);
	const openCount = data.open.reduce((s, p) => s + p.openPositionCount, 0);

	const cards = [
		summaryCard("PnL USD", fmtUsd(total.totalPnlUsd), fmtPct(total.totalPnlPctChange)),
		summaryCard("PnL SOL", fmtSol(total.totalPnlSol), fmtPct(total.totalPnlSolPctChange)),
		summaryCard("Open Balance", fmtUsd(openBalance.toFixed(2)), `${openCount} positions`),
		summaryCard("Unclaimed Fees", fmtUsd(openFees.toFixed(2)), `${data.closed.length} closed`),
	].join("\n");

	const openSection = renderOpen(data.open);
	const closedSection = renderClosed(data.closed);

	return `<section>
<h1>Portfolio</h1>
<div class="cards">${cards}</div>
${openSection}
${closedSection}
</section>`;
}

function renderOpen(pools: readonly OpenPool[]): string {
	if (pools.length === 0) {
		return `<div class="empty">No open positions</div>`;
	}
	const rows = pools.map((p) => {
		const name = `${p.tokenX ?? "?"}/${p.tokenY ?? "?"}`;
		const pnlPct = parseFloat(p.pnlPctChange);
		const rangeBadge = p.outOfRange === true ? badge("OOR", "danger") : badge("in-range", "ok");
		const link = `<a href="${meteoraUrl(p.poolAddress)}" target="_blank" rel="noopener">${escapeHtml(name)}</a>`;
		return `<tr>
<td>${link}</td>
<td>${escapeHtml(String(p.binStep))}</td>
<td>${fmtUsd(p.balances)}</td>
<td>${fmtUsd(p.unclaimedFees)}</td>
<td class="${pnlClass(pnlPct)}">${fmtUsd(p.pnl)} (${fmtPct(pnlPct)})</td>
<td>${rangeBadge}</td>
</tr>`;
	}).join("\n");
	return `<h2>Open Positions (${pools.length})</h2>
${table(["Pool", "Bin", "Balance", "Fees", "PnL", "Range"], rows)}`;
}

function renderClosed(pools: readonly ClosedPool[]): string {
	if (pools.length === 0) {
		return `<div class="empty">No closed positions</div>`;
	}
	const rows = pools.map((p) => {
		const name = `${p.tokenX ?? "?"}/${p.tokenY ?? "?"}`;
		const pnlPct = parseFloat(p.pnlPctChange);
		const link = `<a href="${meteoraUrl(p.poolAddress)}" target="_blank" rel="noopener">${escapeHtml(name)}</a>`;
		const closedAt = p.lastClosedAt ? formatNum(p.lastClosedAt) : "-";

<td>${link}</td>
<td>${fmtUsd(p.totalDeposit)}</td>
<td>${fmtUsd(p.totalWithdrawal)}</td>
<td>${fmtUsd(p.totalFee)}</td>
<td class="${pnlClass(pnlPct)}">${fmtUsd(p.pnlUsd)} (${fmtPct(pnlPct)})</td>
<td class="${pnlClass(pnlPct)}">${fmtSol(p.pnlSol)}</td>
</tr>`;
	}).join("\n");

${table(["Pool", "Deposit", "Withdraw", "Fees", "PnL USD", "PnL SOL"], rows)}`;
}
```

```ts
export const portfolioContent: Effect.Effect<
	string,
	never,
	AppConfig | MeteoraApi | Dlmm
> = Effect.gen(function* () {
	const config = yield* AppConfig;
	const wallet = yield* config.wallet();
	const api = yield* MeteoraApi;
	const dlmm = yield* Dlmm;

	const open = yield* api
		.openPortfolio(wallet, 1, 10)
		.pipe(
			Effect.flatMap((res) => api.enrichOpenPortfolioPnl(res.pools, wallet, { withRanges: true })),
			Effect.flatMap((enriched) => dlmm.attachLivePositions(enriched, wallet)),
			Effect.catchAll(() => Effect.succeed([] as OpenPool[])),
		);

	const closed = yield* api
		.closedPortfolio(wallet, 1, 10)
		.pipe(
			Effect.map((res) => res.pools),
			Effect.catchAll(() => Effect.succeed([] as ClosedPool[])),
		);

	const total = yield* api
		.totalPnl(wallet)
		.pipe(Effect.catchAll(() => Effect.succeed({ totalPnlUsd: "-", totalPnlSol: "-", totalPnlPctChange: "-", totalPnlSolPctChange: "-" } as PortfolioTotal)));

	return renderPortfolio({ total, open, closed });
}).pipe(Effect.catchAll((e) => Effect.succeed(errorBanner(errorMessage(e)))));
```

```ts
import { Effect } from "effect";
import {
	type AgentJournalEntry,
	type JournalCandidate,
	readJournalAll,
} from "../../telegram/agent/journal.js";
import { type AgentState, loadState } from "../../telegram/agent/state.js";
import { errorBanner, escapeHtml } from "../layout.js";
import {
	badge,
	sparkline,
	solscanUrl,
	summaryCard,
	table,
	tsLocal,
} from "../templates.js";

export interface AgentStats {
	readonly cycles: number;
	readonly opens: number;
	readonly holds: number;
	readonly blocked: number;
	readonly tpSl: number;
	readonly closes: number;
	readonly failed: number;
	readonly successRate: number;
}

export function agentStats(entries: readonly AgentJournalEntry[]): AgentStats {
	let opens = 0;
	let holds = 0;
	let blocked = 0;
	let tpSl = 0;
	let closes = 0;
	let failed = 0;
	for (const entry of entries) {
		for (const c of entry.candidates) {
			switch (c.action) {
				case "open":
					opens += 1;
					break;
				case "hold":
					holds += 1;
					break;
				case "tp":
				case "sl":
					tpSl += 1;
					break;
				case "close":
					closes += 1;
					break;
			}
			if (c.guardrail === "blocked") blocked += 1;
			if (c.execution === "failed") failed += 1;
		}
	}
	const decided = opens + holds;
	const successRate = decided > 0 ? Math.round((opens / decided) * 100) : 0;
	return {
		cycles: entries.length,
		opens,
		holds,
		blocked,
		tpSl,
		closes,
		failed,
		successRate,
	};
}

function actionBadge(c: JournalCandidate): string {
	const kind =
		c.action === "open" ? "ok" : c.action === "hold" ? "warn" : "neutral";
	return badge(c.action, kind);
}

function guardrailBadge(c: JournalCandidate): string {
	if (c.guardrail === "blocked") return badge("blocked", "danger");
	return badge("pass", "ok");
}

function executionText(c: JournalCandidate): string {
	if (c.execution === "failed") return badge("failed", "danger");
	if (c.execution === "ok" && c.txSignature) {
		return `<a href="${solscanUrl(c.txSignature)}" target="_blank" rel="noopener" class="mono">${escapeHtml(c.txSignature.slice(0, 12))}…</a>`;
	}
	return "-";
}

export function renderAgent(
	journal: readonly AgentJournalEntry[],
	state: AgentState | null,
): string {
	const stats = agentStats(journal);
	const runningBadge = state?.running === true ? badge("running", "ok") : badge("stopped", "neutral");
	const cards = [
		summaryCard("Cycles", String(stats.cycles), `cycle ${state?.cycle ?? 0}`),
		summaryCard("Opens", String(stats.opens), `${stats.successRate}% success`),
		summaryCard("Holds", String(stats.holds), `${stats.blocked} blocked`),
		summaryCard("TP/SL", String(stats.tpSl), `${stats.closes} closes`),
	].join("\n");

	const opensPerCycle = journal.map(
		(e) => e.candidates.filter((c) => c.action === "open" && c.execution === "ok").length,
	);
	const spark = sparkline(opensPerCycle);

	const journalSection =
		journal.length === 0
			? `<div class="empty">No journal entries</div>`
			: renderJournalTable(journal);

	return `<section>
<h1>Agent ${runningBadge}</h1>
<div class="cards">${cards}</div>
${spark ? `<h2>Opens per cycle</h2>${spark}` : ""}
<h2>Journal</h2>
${journalSection}
</section>`;
}

function renderJournalTable(journal: readonly AgentJournalEntry[]): string {
	const rows = [];
	for (const e of journal) {
		if (e.candidates.length === 0) {
			rows.push(`<tr><td>#${e.cycle}</td><td>${escapeHtml(tsLocal(e.ts))}</td><td colspan="5">no candidates</td></tr>`);
			continue;
		}
		for (const c of e.candidates) {
			const reason = c.blockedReason
				? `<div class="sub">${escapeHtml(c.blockedReason)}</div>`
				: "";
			rows.push(`<tr>
<td>#${e.cycle}</td>
<td>${escapeHtml(tsLocal(e.ts))}</td>
<td>${escapeHtml(c.poolName || c.pool)}</td>
<td>${actionBadge(c)}</td>
<td>${guardrailBadge(c)}${reason}</td>
<td>${executionText(c)}</td>
</tr>`);
		}
	}
	return table(["Cycle", "Time", "Pool", "Action", "Guardrail", "Execution"], rows);
}
```

```ts
		Effect.gen(function* () {
```

```ts
		const req = yield* HttpServerRequest.HttpServerRequest;
		if (req.method !== "POST") {
			return HttpServerResponse.html(loginPage());
		}
		const form = yield* HttpServerRequest.schemaBodyUrlParams(
```

The main entry point:

```ts
function main(): void {
	const program = Effect.gen(function* () {
		const config = yield* AppConfig;
		const cfg = yield* config.get;
		const web = resolveWebConfig(cfg);
		if (!web.enabled) {
			console.log(
				"[web] Web UI disabled (web.enabled=false). Set web.enabled=true in vexis.config.json to start it.",
			);
			return;
		}
		if (!web.password) {
			console.error(
				"[web] Web UI enabled but no password set. Add web.password to vexis.config.json or set VEXIS_WEB_PASSWORD.",
			);
			process.exitCode = 1;
			return;
		}
		const app = buildRouter(web.password);
		yield* HttpServer.serveEffect(app).pipe(
			Effect.provide(
				NodeHttpServer.layer(() => createServer(), { port: web.port }),
			),
		);
	});

	Effect.runFork(
		Effect.scoped(program).pipe(
			Effect.provide(AppLayer),
			Effect.catchAll((e) =>
				Effect.sync(() => {
					console.error("[web] Failed to start:", e);
					process.exitCode = 1;
				}),
			),
		),
	);
}

main();
```
