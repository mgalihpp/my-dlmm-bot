# AI Agent (Autonomous Trading) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a full-autonomous trading agent to the Vexis DLMM bot that screens pools, gets LLM advisory signals, and opens/closes/TP-SLs positions inside configured risk guardrails — with an audit journal and verbose Telegram notifications.

**Architecture:** New `src/telegram/agent/` module following the existing `alerts.ts`/`tpsl.ts` fiber pattern. Pure/testable logic lives in dedicated modules (`heuristic`, `decision`, `guardrails`, `params`, `journal`, `state`, `llm`, `format`); `engine.ts` orchestrates the run loop + executor using existing `fx.ts` helpers; `commands.ts` registers `/agent` commands. No new npm dependencies — HTTP via Node 20 built-in `fetch`.

**Tech Stack:** TypeScript, Effect (functional effects + layer DI), grammY (Telegram), vitest (tests), Meteora services (`fx.ts` façade: `screenPools`, `api`, `dlmm`, `zap`).

## Global Constraints

- Follow existing file conventions: `src/` modules import Effect services via `fx.ts` façade functions; pure logic has no I/O.
- Config type additions go in `src/domain/config.ts` (plain TS interfaces) and `src/services/Config.ts` (resolvers, pattern: `resolveCreatePresetFrom`).
- JSON state files live in `process.cwd()` and are git-ignored (add `.vexis-agent.json`, `.vexis-agent-journal.json`).
- No new npm dependencies. No comments in code unless absolutely needed.
- Tests live in `test/*.test.ts` and run via `npx vitest run <file>`; typecheck via `npm run typecheck`; format via `npm run format`.
- Every pure decision function is unit-tested. Engine/commands are wired end-to-end and verified via `npm run typecheck` + `npm run bot` smoke.
- See spec: `docs/superpowers/specs/2026-08-08-ai-agent-design.md`.

import type { ScreenedPool } from "../../domain/screened.js";

const clamp = (v: number, lo = 0, hi = 1) => Math.min(hi, Math.max(lo, v));

/** Deterministic 0-100 quality score from discovery metrics. */
export function heuristicScore(pool: ScreenedPool): number {
	const feeTvl = clamp(pool.feeActiveTvlRatio / 0.05);
	const organic = clamp(pool.organicScore / 100);
	const holders = clamp(pool.holders / 1000);
	const volume = clamp(pool.volume / 100_000);
	const binStep = clamp(1 - pool.binStep / 125);
	const raw =
		100 * (0.35 * feeTvl + 0.25 * organic + 0.1 * holders + 0.1 * volume + 0.2 * binStep);
	return Math.round(clamp(raw, 0, 100));
}

export function rankPools(
	pools: readonly ScreenedPool[],
	opts: { minCandidate: number; maxCandidates: number },
): ScreenedPool[] {
	return pools
		.map((p) => ({ p, h: heuristicScore(p) }))
		.sort((a, b) => b.h - a.h)
		.filter((r) => r.h >= opts.minCandidate)
		.slice(0, opts.maxCandidates)
		.map((r) => r.p);
}
```

import type { CreatePreset } from "../../domain/config.js";
import type { CreatePositionParams, StrategyType } from "../../domain/onchain.js";

const DEFAULT_SOL_BINS = { minBin: -69, maxBin: 0 } as const;

export function buildCreateParams(input: {
	poolAddress: string;
	strategy: StrategyType;
	range: CreatePreset["range"];
	amountSol: number;
}): CreatePositionParams {
	const r = input.range ?? { type: "default" };
	const base = {
		poolAddress: input.poolAddress,
		strategy: input.strategy,
		totalXAmount: "0",
		totalYAmount: String(input.amountSol),
		singleSidedX: false,
		singleSidedY: true,
		amountsAreHuman: true,
	};
	if (r.type === "pct" && r.minPct != null && r.maxPct != null) {
		return { ...base, minPct: r.minPct, maxPct: r.maxPct };
	}
	if (r.type === "bin" && r.minBin != null && r.maxBin != null) {
		return { ...base, minBinId: r.minBin, maxBinId: r.maxBin, relativeBins: true };
	}
	return {
		...base,
		minBinId: DEFAULT_BINS.minBin,
		maxBinId: DEFAULT_BINS.maxBin,
		relativeBins: true,
	};
}
```

```ts
import type { ResolvedAgentConfig } from "../../services/Config.js";

export interface LlmCandidate {
	id: string;
	pair: string;
	heuristic: number;
	feeActiveTvlRatio: number;
	organicScore: number;
	holders: number;
	volume: number;
}

export interface LlmSignal {
	pool: string;
	favorability: number;
	rationale: string;
}

const clampFav = (v: unknown): number | null => {
	if (typeof v !== "number" || !Number.isFinite(v)) return null;
	return Math.max(-1, Math.min(1, v));
};

export function buildPrompt(candidates: readonly LlmCandidate[]): string {
	const table = candidates
		.map(
			(c) =>
				`- pool=${c.id} pair=${c.pair} heuristic=${c.heuristic} feeTvlRatio=${c.feeActiveTvlRatio.toFixed(4)} organic=${c.organicScore} holders=${c.holders} volume=${c.volume}`,
		)
		.join("\n");
	return [
		"You are a market advisor for a DLMM liquidity bot. For each pool, output a `favorability` score between -1 (strongly avoid) and +1 (strongly favorable), plus a one-line rationale.",
		"Reply with a JSON array only, never markdown: [{\"pool\":\"<exact pool id>\",\"favorability\":0.5,\"rationale\":\"...\"}]",
		"",
		"Candidates:",
		table,
	].join("\n");
}

export function parseLlmResponse(content: string): LlmSignal[] {
	const cleaned = content.trim().replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/, "");
	let parsed: unknown;
	try {
		parsed = JSON.parse(cleaned);
	} catch {
		return [];
	}
	const arr = Array.isArray(parsed) ? parsed : (parsed as { candidates?: unknown }).candidates;
	if (!Array.isArray(arr)) return [];
	const out: LlmSignal[] = [];
	for (const item of arr) {
		const o = item as { pool?: unknown; favorability?: unknown; rationale?: unknown };
		if (typeof o.pool !== "string" || o.pool === "") continue;
		const fav = clampFav(o.favorability);
		if (fav === null) continue;
		out.push({
			pool: o.pool,
			favorability: fav,
			rationale: typeof o.rationale === "string" ? o.rationale : "",
		});
	}
	return out;
}

export async function requestSignals(opts: {
	cfg: ResolvedAgentConfig;
	candidates: readonly LlmCandidate[];
}): Promise<{ signals: LlmSignal[]; degraded: boolean }> {
	const { cfg } = opts;
	if (!cfg.llm.apiKey || opts.candidates.length === 0) {
		return { signals: [], degraded: true };
	}
	const controller = new AbortController();
	const timer = setTimeout(() => controller.abort(), cfg.llm.timeoutMs);
	try {
		const res = await fetch(`${cfg.llm.baseUrl}/chat/completions`, {
			method: "POST",
			headers: {
				"Content-Type": "application/json",
				Authorization: `Bearer ${cfg.llm.apiKey}`,
			},
			signal: controller.signal,
			body: JSON.stringify({
				model: cfg.llm.model,
				messages: [{ role: "user", content: buildPrompt(opts.candidates) }],
				temperature: 0,
			}),
		});
		if (!res.ok) return { signals: [], degraded: true };
		const body = (await res.json()) as { choices?: { message?: { content?: string } }[] };
		const content = body.choices?.[0]?.message?.content;
		if (!content) return { signals: [], degraded: true };
		return { signals: parseLlmResponse(content), degraded: false };
	} catch {
		// timeout / network: degrade to heuristic-only
		return { signals: [], degraded: true };
	} finally {
		clearTimeout(timer);
	}
}
```

```ts
import type { ResolvedAgentConfig } from "../../services/Config.js";

export interface GuardGuardOk {
	ok: boolean;
	reason: string | null;
}

export function checkOpenGuardrail(input: {
	amountSol: number;
	deployedSol: number;
	maxSolPerPosition: number;
	maxTotalSol: number;
	maxOpenPositions: number;
	openPositionCount?: number;
}): GuardGuardOk {
	if (input.amountSol > input.maxSolPerPosition + 1e-9) {
		return { ok: false, reason: `amount ${input.amountSol} > per-position cap ${input.maxSolPerPosition}` };
	}
	if (input.deployedSol + input.amountSol > input.maxTotalSol + 1e-9) {
		return { ok: false, reason: `total ${input.deployedSol + input.amountSol} > cap ${input.maxTotalSol}` };
	}
	if (input.openPositionCount !== undefined && input.openPositionCount >= input.maxOpenPositions) {
		return { ok: false, reason: `already ${input.openPositionCount} open positions` };
	}
	return { ok: true, reason: null };
}

export function checkCooldown(input: {
	lastExecutionAt: number | null;
	nowMs: number;
	txCooldownMs: number;
}): GuardGuardOk {
	if (input.lastExecutionAt == null) return { ok: true, reason: null };
	if (input.nowMs - input.lastExecutionAt < input.txCooldownMs) {
		return { ok: false, reason: "within tx cooldown" };
	}
	return { ok: true, reason: null };
}

/** SOL amount to deploy for one new position, bounded by the remaining budget. */
export function deriveOpenAmount(deployedSol: number, cfg: ResolvedAgentConfig): number {
	const wanted = Math.min(cfg.maxSolPerPosition, Math.max(0, cfg.maxTotalSol - deployedSol));
	return Math.max(0, Math.round(wanted * 1000) / 1000);
}

import type { ScreenedPool } from "../../domain/screened.js";
import { heuristicScore } from "./heuristic.js";
import type { LlmSignal } from "./llm.js";

export type AgentAction = "open" | "hold";

export interface CandidateDecision {
	pool: ScreenedPool;
	heuristicScore: number;
	favorability: number | null;
	rationale: string | null;
	score: number;
	action: AgentAction;
}

const OPEN_THRESHOLD_MIN = 0.2;

/** favorability[-1,1] → merged 0-100, heuristic has 80% weight. */
export function combineScore(heuristic: number, favorability: number | null): number {
	if (favorability == null) return heuristic;
	const favFrame = ((favorability + 1) / 2) * 100;
	return Math.round(0.8 * heuristic + 0.2 * favFrame);
}

export function decideCandidates(input: {
	pools: readonly ScreenedPool[];
	signals: readonly LlmSignal[];
	minScoreToOpen: number;
	candidatesLimit?: number;
}): CandidateDecision[] {
	const sigMap = new Map<string, LlmSignal>(input.signals.map((s) => [s.pool, s]));
	const limit = input.candidatesLimit ?? input.pools.length;

	const decisions: CandidateDecision[] = input.pools
		.slice(0, limit)
		.map((pool) => {
			const h = heuristicScore(pool);
			const sig = sigMap.get(pool.pool);
			const favorability = sig?.favorability ?? null;
			const score = combineScore(h, favorability);
			const threshold = input.minScoreOpen;
			const action: AgentAction = score >= threshold ? "open" : "hold";
			return {
				pool,
				heuristicScore: h,
				favorability,
				rationale: sig?.rationale ?? null,
				score,
				action,
			};
		});

	const opens = decisions.filter((d) => d.action === "open");
	if (opens.length === 0) return decisions;
	opens.sort((a, b) => b.score - a.score);
	const cap = opens.length;
	return [
		...opens,
		...decisions.filter((d) => d.action !== "open").map((d) => ({ ...d, action: ("hold" as const), score: d.score - favThresholdHint })),
	];
}

export function readJournal(n = 10, file = DEFAULT_FILE): AgentJournalEntry[] {
	if (!existsSync(file)) return [];
	try {
		const lines = readFileSync(file, "utf8").split("\n").filter(Boolean);
		const parsed: AgentJournalEntry[] = [];
		for (const line of lines.slice(-Math.max(1, n))) {
			try {
				parsed.push(JSON.parse(line) as AgentJournalEntry);
			} catch {}
		}
		return parsed;
	} catch {
		return [];
	}
}

export function formatStatus(state: AgentState, cfg: ResolvedAgentConfig): string {

	].join("\n");
	return lines;

export function formatCycleSummary(entries: readonly AgentJournalEntry[], degraded: boolean): string {
	const last = entries[0];
	if (!last) return "🤖 No agent cycle has run yet.";
	const lines = [tgBold(`🤖 Agent cycle #${last.cycle}`)];
	if (degraded) lines.push("⚠️ LLM degraded — heuristic only");
	for (const c of last.candidates) {
		const sign = c.favorability == null ? "—" : c.favorability.toFixed(2);
		const action = c.action.toUpperCase();
		const executed = c.execution === "ok" ? ` [${tgCode(c.txSignature ?? "")}] ` : c.execution === "failed" ? " ❌FAILED" : c.guardrail === "blocked" ? ` ⛔blocked: ${escapeMarkdown(c.blockedReason ?? "")}` : "";
		lines.push(`${action} ${escapeMarkdown(c.poolName)} score ${c.score} fav ${sign}${executed}`);
	}
	return lines.join("\n");
}

export function formatJournal(entries: readonly AgentJournalEntry[], n: number): string {
	const lines = [tgBold(`📒 Agent journal (last ${n})`)];
	for (const e of entries.slice(0, n)) {
		const opened = e.candidates.filter((c) => c.execution === "ok").length;
		const blocked = e.candidates.filter((c) => c.guardrail === "blocked").length;
		lines.push(
			`• #${e.cycle} ${escapeMarkdown(e.ts)} llm=${escapeMarkdown(e.llmStatus)} opened=${opened} blocked=${blocked}`,
		);
	}
	return lines.join("\n");
}

- Produces: `RuntimeAgent { state: AgentState; start(); stop(); runCycle(): Promise<void> }`, `createAgent(bot: Bot, chatId: string): RuntimeAgent`, `formatCycleExecutor` internals.

Design: two fibers — a full decision `intervalMinutes` loop and a 60-second "event" fiber that re-checks TP/SL on agent-managed positions and reacts to out-of-range. Both share one `runCycle` that uses a `state.running` lock. `runCycle` writes the journal each run.

import { checkOpenGuardrail, checkCooldown, deriveOpenAmount } from "./guardrails.js";

import { loadState, saveState, type AgentState } from "./state.js";
import { formatCycleSummary, formatStatus, formatJournal } from "./format.js";
import { MD } from "../utils.js";
import { escapeMarkdown } from "../format.js";

export interface RuntimeState extends AgentState {
	deployedSol: number;
}

export interface RuntimeAgent {
	state: RuntimeState;
	start(): void;
	stop(): void;
	runCycle(): Promise<void>;
}

function send(bot: Bot, chatId: string, msg: string) {
	return bot.api.sendMessage(chatId, msg, MD);
}

export function createAgent(bot: Bot, chatId: string, runImmediately = true): RuntimeAgent {
	let intervalFiber: Fiber.RuntimeFiber<unknown, unknown> | null = null;
	let eventFiber: Fiber.RuntimeFiber<unknown, unknown> | null = null;

	const apiState = {
		...loadState(),
		deployedSol: 0,
	} as RuntimeState;

	const stopFiber = (f: Fiber.RuntimeFiber<unknown, unknown> | null) => {
		if (f) runtime.runFork(Fiber.interrupt(f));
	};

	const schedule = (label: string, intervalMs: number, job: () => Promise<void>): Fiber => {
		return runtime.runFork(
			Effect.tryPromise(job).pipe(
				Effect.catchAll((e) => Effect.sync(() => console.error(`[agent] ${label} failed:`, e))),
				Effect.repeat(Schedule.spaced(Duration.millis(intervalMs))),
			),
		);
	};

	const engine: RuntimeAgent = {
		state: apiState,
		start() {
			stopCycle(intervalFiber);
			stopCycle(eventFiber);
			const cfgEffect = getConfig();
			// bootstrap once for interval config
			void cfgEffect.then((cfg) => {
				const agentCfg = resolveAgentConfigFrom(cfg);
				this.state.enabled = true;
				saveState(this.state);
				intervalFiber = schedule(
					"loop",
					agentCfg.intervalMinutes * 60_000,
					() => runCycle(this),
				);
				eventFiber = schedule("event", 60_000, () => runCycle(this));
			});
		},
		stop() {
			stopCycle(intervalFiber);
			stopCycle(eventFiber);
			intervalFiber = null;
			eventFiber = null;
			this.state.enabled = false;
			this.state.running = false;
			saveState(this.state);
		},
		async runCycle() {
			if (this.state.running || !this.state.enabled) return;
			this.state.running = true;
			try {
				const cfg = resolveAgentConfigFrom(await getConfig());
				const wallet = await resolveWallet();
				const open = await api.openPortfolio(wallet, 1, 100);
				const deployed = Number(open.total?.balancesSol ?? 0);
				this.state.deployedSol = deployed;
				const pools = await this.evaluatePlansSync(cfg, open.pools, wallet, bot);
				// see below: evaluatePlansSync appends journal; returns summary
			} catch (e) {
				console.error("[agent] cycle error:", e);
			} finally {
				this.state.running = false;
				this.state.lastCycleAt = new Date().toISOString();
				saveState(this.state);
			}
		},
	};
```

async function evaluatePlans(
	rt: RuntimeAgent,
	bot: Bot,
	chatId: string,
	cfg: ReturnType<typeof resolveAgentConfigFrom>,
) {
	const journal: AgentJournalEntry = {
		ts: new Date().toISOString(),
		cycle: ++rt.state.cycle,
		llmStatus: "skipped",
		candidates: [],
	};

	// 1. TP/SL + event pass over managed positions
	// 2. Screening + candidate decisions
	const screen = await screenPools();
	const ranked = rankPools(screen.pools, {
		minCandidate: cfg.minCandidate,
		maxCandidates: cfg.maxCandidates,
	});
	const llmCandidates: LlmCandidate[] = ranked.map((p) => ({
		id: p.pool,
		pair: `${p.baseSymbol}/${p.quoteSymbol}`,
		heuristic: heuristicScore(p),
		feeActiveTvlRatio: p.feeActiveTvlRatio,
		organicScore: p.organicScore,
		holders: p.holders,
		volume: p.volume,
	}));
	const { signals, degraded } = await requestSignals({ cfg, candidates: llmCandidates });
	if (degraded) journal.llmStatus = cfg.llm.apiKey ? "degraded" : "skipped";
	else journal.llmStatus = "ok";

	const decisions = decideCandidates({
		pools: ranked,
		signals,
		minScoreOpen: cfg.minCandidate,
		candidatesLimit: cfg.maxCandidates,
	});

	const pending: Array<CandidateDecision> = decisions.filter((d) => d.action === "open");
	const lastExec = rt.state.executions[rt.state.executions.length - 1]?.at ?? null;

	for (const d of pending) {
		// guardrail verification
		if (!checkOpenGuardrail({ amountSol: 0, ... }).ok) {
			journalBlocked(journal, d, "cap");
			continue;
		}
		const amountSol = deriveOpenAmount(rt.state.deployedSol, cfg);
		if (!(amountSol > 0)) { blocked(journal, d, "no budget"); continue; }
		if (!checkCooldown({ lastExecutionAt: lastExec?. as number .. now })) {
			journal blocked "cooldown"; continue;
		}
		// open it
		const preset = getCreatePreset();
		const params = buildCreateParams({ poolAddress: d.pool.pool, preset, amountSol });
		try {
			const res = await dlmm.createPosition(params);
			const sig = res.signatures.join(",");
			rt.state.plans.push({ pool, poolAddress, amountSol, positionAddress: res.positions[0] ?? null, openedAt: new Date().toISOString() });
			rt.state.executions.push({ at: new Date().toISOString(), action: "open", pool: d.pool.pool, txSignature: sig });
			journal.candidates.push(okEntry(d, "ok", sig));
			appendJournal(journal);
		} catch (e) {
			journal.candidates.push({ ..., execution: "failed" });
			appendJournal(journal);
		}
	}

	await send(bot, chatId, formatCycleSummary(readJournal(1), degraded));
	saveState(rt.state);
}

```ts
async function evaluateTpSl(rt, bot, chatId, cfg, wallet, openPools) {
	for (const plan of rt.state.plans) {
		if (!plan.positionAddress) continue;
		const poolPools = open.find((p) => p.poolAddress === plan.pool);
		if (!poolPools) continue; // no longer open — already closed
		const pdata = await api.positionPnl(plan.pool, wallet, "open");
		const pos = pdata.positions.find((pp) => pp.positionAddress === plan.positionAddress);
		if (!pos || pos.isClosed) continue;
		const pct = pos.pnlSolPctChange != null ? parseFloat(String(pos.pnlSolPctChange)) : pos.pnlPctChange;
		const action = tpslAction(pct, cfg.tpPct, cfg.slPct);
		if (action === "tp" || action === "sl") {
			try {
				const out = await zap.closeAndZapOut(rt.state.pool, plan.positionAddress);
				const sig = out.closeSig ?? out.zapSig ?? out.account ?? "";
				rt.state.plans = rt.state.plans.filter((x) => x.positionAddress !== plan.positionAddress);
				rt.state.executions.push({ at: new Date().toISOString(), action, pool: plan.pool, txSignature: sig });
				rt.state.deployedSol = 0; // recompute next cycle
				appendJournal({ ts, cycle: rt.state.cycle, llmStatus: rt.state.llmStatus, candidates: [{ ...plan, action, guardrail:"pass", execution:"ok", txSignature: sig }] });
				await send(bot, chatId, formatCycleSummary(...) ...);
			} catch (e) {
				console.error("[agent] tp/sl close failed:", e);
			}
		}
	}
}

import type { Bot } from "grammy";
import { escapeMarkdown, tgBold } from "../format.js";
import { MD } from "../utils.js";
import type { RuntimeAgent } from "./engine.js";
import { formatStatus, formatJournal } from "./format.js";
import { readJournal } from "./journal.js";
import { resolveAgentConfigFrom } from "../../services/Config.js";
import { getConfig } from "../fx.js";

export function registerAgentCommands(bot: Bot, rt: RuntimeAgent) {
	bot.command("agent", async (ctx) => {
		const [cmd, arg] = (ctx.match as string).trim().split(/\s+/);
		switch (cmd) {
			case "start": {
				rt.start();
				await ctx.reply("🤖 Agent started.", MD);
				break;
			}
			case "stop": {
				rt.stop();
				await ctx.reply("🛑 Agent stopped.", MD);
				break;
			}
			case "status": {
				const cfg = resolveAgentConfigFrom(await getConfig());
				await ctx.reply(formatStatus(rt.state as any as AgentState, cfg), MD);
				break;
			}
			case "journal": {
				const n = Math.min(parseInt(arg || "5", 10) || 5, 20);
				await ctx.reply(formatJournal(readJournal(n), n), MD);
				break;
			}
			default: {
				const lines = [
					tgBold("🤖 AI Agent"),
					escapeMarkdown("/agent start — enable & loop"),
					escapeMarkdown("/agent stop — emergency stop"),
					escapeMarkdown("/agent status — state & caps"),
					escapeMarkdown("/agent journal [n] — last decisions"),
				];
				await ctx.reply(lines.join("\n"), MD);
			}
		}
	});
}
```
